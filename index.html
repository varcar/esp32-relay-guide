BEGINNER PROJECT

Control a relay
from your phone.

A simple ESP32 project with a web page that has ON and OFF buttons. No extra apps needed.

1. What you need

ESP32 Development board
Relay module 1 channel
Jumper wires 3 wires
USB cable For uploading

2. Connect the wires

*Connect VCC to the voltage specified on your relay module (3.3V or 5V). Some 5V modules do not work reliably with ESP32's 3.3V control signal.

Test with a low-voltage circuit first. Do not connect household AC mains wiring as a beginner.

3. Upload the code

  1. Open Arduino IDE and select your ESP32 board.
  2. Create a new sketch. Copy the entire code below and paste it in.
  3. Change YOUR_WIFI_NAME and YOUR_WIFI_PASSWORD.
  4. Click Upload.
  5. Open Serial Monitor at 115200 baud. Copy the IP address shown.
  6. Connect your phone to the same Wi-Fi. Open http://YOUR_ESP32_IP in the browser. Tap ON / OFF.

This controls the relay over your local Wi-Fi network, not worldwide.

4. Complete Arduino code

Everything you need is here. Copy it into Arduino IDE.

ESP32_Relay_Lite.ino
#include <WiFi.h>
#include <WebServer.h>

// -------- EDIT ONLY THESE SETTINGS --------
const char* WIFI_SSID = "YOUR_WIFI_NAME";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";
constexpr uint8_t RELAY_PIN = 26;
constexpr bool RELAY_ACTIVE_LOW = true; // Most relay modules use LOW = ON
constexpr bool TURN_OFF_IF_WIFI_LOST = true; // Safety: OFF if Wi-Fi disconnects
// ------------------------------------------

WebServer server(80);
bool relayOn = false;
unsigned long lastReconnectAttempt = 0;
unsigned long wifiDisconnectedAt = 0;

void applyRelay(bool on) {
  digitalWrite(RELAY_PIN, (on == RELAY_ACTIVE_LOW) ? LOW : HIGH);
  relayOn = on;
  Serial.printf("Relay: %s\n", on ? "ON" : "OFF");
}

void statusReply() {
  server.sendHeader("Cache-Control", "no-store, no-cache, must-revalidate");
  server.sendHeader("Pragma", "no-cache");
  server.send(200, "application/json", relayOn ? "{\"on\":true}" : "{\"on\":false}");
}

void handleSet() {
  if (!server.hasArg("state")) { server.send(400, "text/plain", "Missing state"); return; }
  const String requested = server.arg("state");
  if (requested == "on") applyRelay(true);
  else if (requested == "off") applyRelay(false);
  else { server.send(400, "text/plain", "Invalid state"); return; }
  statusReply();
}

void setup() {
  // Set output OFF before configuring the pin. Relay must have a reliable supply.
  digitalWrite(RELAY_PIN, RELAY_ACTIVE_LOW ? HIGH : LOW);
  pinMode(RELAY_PIN, OUTPUT);
  applyRelay(false);

  Serial.begin(115200);
  delay(100);
  Serial.println("\nESP32 Relay Lite");
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false); // Reduces web server latency on many ESP32 boards
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  server.on("/", HTTP_GET, []() {
    server.sendHeader("Cache-Control", "no-store");
    server.send_P(200, "text/html", HTML);
  });
  server.on("/api/status", HTTP_GET, statusReply);
  server.on("/api/set", HTTP_POST, handleSet);
  server.onNotFound([](){server.send(404, "text/plain", "Not found");});
  server.begin();
  Serial.println("Wi-Fi connecting. Watch for IP address...");
}

void loop() {
  server.handleClient();
  const unsigned long now = millis();
  if (WiFi.status() == WL_CONNECTED) {
    wifiDisconnectedAt = 0;
    static bool announced = false;
    static IPAddress lastIP;
    if (!announced || WiFi.localIP() != lastIP) {
      Serial.print("Open http://"); Serial.println(WiFi.localIP());
      lastIP = WiFi.localIP(); announced = true;
    }
  } else {
    if (wifiDisconnectedAt == 0) wifiDisconnectedAt = now;
    if (TURN_OFF_IF_WIFI_LOST && relayOn && (now - wifiDisconnectedAt >= 3000UL)) applyRelay(false);
    if (now - lastReconnectAttempt >= 10000UL) {
      lastReconnectAttempt = now;
      Serial.println("Wi-Fi disconnected; reconnecting...");
      WiFi.reconnect();
    }
  }
  delay(2); // Yield to Wi-Fi tasks; never block the web server for seconds.
}

Relay connected to GPIO 26. If ON and OFF work backward, change RELAY_ACTIVE_LOW to false.

Not working?

Page won't open? Check that your phone is on the same Wi-Fi and use http://, not https://.

Relay doesn't click? Check GPIO 26, GND, relay voltage rating, and whether your module accepts 3.3V input signals.

Code won't upload? Try holding the ESP32 BOOT button while uploading.
